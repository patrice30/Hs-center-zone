function doGet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetM = ss.getSheetByName("MEMBRES"); var sheetD = ss.getSheetByName("DEPOTS"); var membres = []; if (sheetM && sheetM.getLastRow() > 1) { var dataM = sheetM.getRange(2, 1, sheetM.getLastRow() - 1, 4).getValues(); membres = dataM.map(function(row) { return { id: row[0], nom: row[1], tel: row[2], servi: row[3] === true || row[3] === "true" }; }); } var depots = []; if (sheetD && sheetD.getLastRow() > 1) { var dataD = sheetD.getRange(2, 1, sheetD.getLastRow() - 1, 5).getValues(); depots = dataD.map(function(row) { return { date: row[0], nom: row[1], montant: row[2], operateur: row[3], ref: row[4] }; }); } var output = JSON.stringify({ membres: membres, depots: depots }); return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === "addMembre") { var sheetM = ss.getSheetByName("MEMBRES"); sheetM.appendRow([data.id, data.nom, data.tel, false]); } else if (data.action === "addDepot") { var sheetD = ss.getSheetByName("DEPOTS"); sheetD.appendRow([data.date, data.nom, data.montant, data.operateur, data.ref]); } else if (data.action === "updateServi") { var sheetM = ss.getSheetByName("MEMBRES"); var values = sheetM.getRange(2, 1, sheetM.getLastRow() - 1, 1).getValues(); for (var i = 0; i < values.length; i++) { if (values[i][0] == data.id) { sheetM.getRange(i + 2, 4).setValue(true); break; } } } else if (data.action === "deleteMembre") { var sheetM = ss.getSheetByName("MEMBRES"); var values = sheetM.getRange(2, 1, sheetM.getLastRow() - 1, 1).getValues(); for (var i = 0; i < values.length; i++) { if (values[i][0] == data.id) { sheetM.deleteRow(i + 2); break; } } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); }
